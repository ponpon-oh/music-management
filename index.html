Google Apps Script (Code.gs) の doPost 関数だけを、以下の内容に置き換えてください。 （doGet や他の関数（out_, json_, props_, dataFile_, photoFolder_, savePhoto_）はそのままで大丈夫です） 理由： これまでは「保存した端末の時計の時刻」を版数(rev)として使っていました。 そのため、端末の時計が少しでもズレていると「サーバーの方が新しいはずなのに、この端末の方が新しいと誤認して同期済みと表示される」ことが起こり得ました。 この修正で、版数をサーバー側で必ず1つずつ増える形で発行するようにし、この種の誤判定を根本的に防ぎます。 ------------------------------------------------------------ function doPost(e){ var p = (e && e.parameter) || {}; var action=p.action, secret=p.secret, data=p.data, name=p.name; if (!action && e && e.postData && e.postData.contents){ try { var o=JSON.parse(e.postData.contents); action=o.action; secret=o.secret; data=o.data; name=o.name; } catch(_){} } if (secret !== SECRET) return json_({error:'auth'}); if (action === 'save') { var parsed; try { parsed = JSON.parse(data); } catch(err){ return json_({error:'invalid_json'}); } if (!parsed || typeof parsed !== 'object' || !Array.isArray(parsed.items)) { return json_({error:'invalid_shape'}); } var f = dataFile_(false); var curRev = 0; if (f) { try { var cur = JSON.parse(f.getBlob().getDataAsString()); curRev = cur.rev || 0; } catch(_){} } var newRev = Math.max(curRev + 1, Date.now()); parsed.rev = newRev; dataFile_(true).setContent(JSON.stringify(parsed)); return json_({ok:true, rev:newRev}); } if (action === 'photo') { return json_({url: savePhoto_(name, data)}); } return json_({ok:false}); } ------------------------------------------------------------ 反映手順（いつもと同じ）： 1. script.google.com を開く 2. Code.gs の doPost 関数だけを上記に置き換えて保存 3. 右上「デプロイ」→「デプロイを管理」→ 既存デプロイの鉛筆アイコン →「新しいバージョン」を選んで「デプロイ」 （URLは変わりません。新規デプロイを作らないよう注意）
